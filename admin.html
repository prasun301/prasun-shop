async function verifySession() { const token = localStorage.getItem("sessionToken"); // If no token exists at all, kick them to login if (!token) { window.location.href = "login.html"; return; } // Ask the Worker if this token is actually real and unexpired const response = await fetch("https://prasun-shop-api.prasun301.workers.dev/verify-session", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ token }) }); const result = await response.json(); // If the Worker says it's fake or expired, delete it and kick them out if (!result.valid) { localStorage.removeItem("sessionToken"); window.location.href = "login.html"; } } // Run this immediately when the page loads verifySession();
