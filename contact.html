/** * ============================================================================ * MULTI-SITE CONTACT BACKEND * ============================================================================ * * Cloudflare Worker * * Supports: * * POST / * * Request formats: * * 1. multipart/form-data * - Used by PRASUN SHOP contact page * - Supports file attachments * * 2. application/json * - Supported for other websites / legacy forms * * Required secret: * * RESEND_API_KEY * * Required variable: * * DESTINATION_EMAIL * * Optional: * * MAX_FILE_SIZE_MB * * ============================================================================ */ "use strict"; /* ============================================================================ CONSTANTS ========================================================================== */ const DEFAULT_MAX_FILE_SIZE = 10 * 1024 * 1024; /* ============================================================================ WORKER ========================================================================== */ export default { async fetch( request, env, ctx ) { const corsHeaders = getCorsHeaders( request ); /* * ------------------------------------------------------------ * CORS PREFLIGHT * ------------------------------------------------------------ */ if ( request.method === "OPTIONS" ) { return new Response( null, { status: 204, headers: corsHeaders } ); } else if ( contentType.includes( "application/x-www-form-urlencoded" ) ) { contactData = await parseUrlEncodedRequest( request ); } else { return jsonResponse( { success: false, error: "Unsupported request content type." }, corsHeaders, 415 ); } /* * -------------------------------------------------------- * SPAM HONEYPOT * -------------------------------------------------------- */ if ( cleanText( contactData?.honeypot ) ) { /* * Deliberately return a generic successful response * so basic bots do not learn that the honeypot fired. */ return jsonResponse( { success: true, message: "Message received." }, corsHeaders ); } /* * -------------------------------------------------------- * NORMALIZE * -------------------------------------------------------- */ const name = cleanText( contactData?.name ); const email = cleanText( contactData?.email ); const siteName = cleanText( contactData?.siteName ) || "Website"; const topic = cleanText( contactData?.topic ); const orderId = cleanText( contactData?.orderId ); const message = cleanText( contactData?.message ); const source = cleanText( contactData?.source ); /* * -------------------------------------------------------- * VALIDATION * -------------------------------------------------------- */ if ( !email || !isValidEmail( email ) ) { return jsonResponse( { success: false, error: "Please provide a valid email address." }, corsHeaders, 400 ); } if ( !message ) { return jsonResponse( { success: false, error: "Message is required." }, corsHeaders, 400 ); } if ( message.length > 5000 ) { return jsonResponse( { success: false, error: "Message is too long. Maximum length is 5000 characters." }, corsHeaders, 400 ); } if ( name.length > 120 ) { return jsonResponse( { success: false, error: "Name is too long." }, corsHeaders, 400 ); } if ( siteName.length > 150 ) { return jsonResponse( { success: false, error: "Site name is too long." }, corsHeaders, 400 ); } /* * -------------------------------------------------------- * EXTRA FIELDS * -------------------------------------------------------- * * This keeps the backend compatible with other sites * that may send additional form fields. */ const extraFields = contactData?.extraFields || {}; /* * -------------------------------------------------------- * BUILD EMAIL TEXT * -------------------------------------------------------- */ const lines = []; lines.push( `Name: ${name || "Not provided"}` ); lines.push( `Email: ${email}` ); lines.push( `Website: ${siteName}` ); if ( source ) { lines.push( `Source: ${source}` ); } if ( topic ) { lines.push( `Topic: ${topic}` ); } if ( orderId ) { lines.push( `Order ID: ${orderId}` ); } /* * Add additional fields. */ for ( const [ key, value ] of Object.entries( extraFields ) ) { if ( !cleanText( value ) ) { continue; } const formattedKey = formatFieldName( key ); /* * Prevent duplicate standard fields. */ if ( [ "Name", "Email", "Site Name", "Website", "Source", "Topic", "Order Id", "Message", "Attachment" ].includes( formattedKey ) ) { continue; } lines.push( `${formattedKey}: ${cleanText(value)}` ); } lines.push(""); lines.push( "Message:" ); lines.push( message ); /* * -------------------------------------------------------- * ATTACHMENT * -------------------------------------------------------- */ const attachment = contactData?.attachment || null; /* * -------------------------------------------------------- * MAX FILE SIZE * -------------------------------------------------------- */ const configuredMaxMB = Number( env?.MAX_FILE_SIZE_MB ); const maxFileSize = Number.isFinite( configuredMaxMB ) && configuredMaxMB > 0 ? ( configuredMaxMB * 1024 * 1024 ) : DEFAULT_MAX_FILE_SIZE; /* * -------------------------------------------------------- * RESEND PAYLOAD * -------------------------------------------------------- */ const resendPayload = { from: `${sanitizeSenderName(siteName)} `, to: [ env.DESTINATION_EMAIL ], subject: `New Message from [${siteName}]`, text: lines.join( "\n" ) }; /* * -------------------------------------------------------- * RESEND ATTACHMENT * -------------------------------------------------------- * * Resend accepts attachment data. * * We encode the uploaded file as Base64. */ if ( attachment ) { if ( !(attachment instanceof File) ) { return jsonResponse( { success: false, error: "Invalid attachment." }, corsHeaders, 400 ); } if ( attachment.size > maxFileSize ) { return jsonResponse( { success: false, error: `Attachment is too large. Maximum allowed size is ${Math.round( maxFileSize / 1024 / 1024 )} MB.` }, corsHeaders, 400 ); } const fileName = sanitizeFileName( attachment.name ); const extension = getFileExtension( fileName ); if ( !ALLOWED_EXTENSIONS.has( extension ) ) { return jsonResponse( { success: false, error: "This attachment type is not supported." }, corsHeaders, 400 ); } const contentType = cleanText( attachment.type ) .toLowerCase(); if ( contentType && !ALLOWED_CONTENT_TYPES.has( contentType ) ) { return jsonResponse( { success: false, error: "The attachment content type is not supported." }, corsHeaders, 400 ); /* * Add attachment information to email body. */ resendPayload.text += `\n\nAttachment: ${fileName}`; } /* * -------------------------------------------------------- * SEND THROUGH RESEND * -------------------------------------------------------- */ const resendResponse = await fetch( "https://api.resend.com/emails", { method: "POST", headers: { "Authorization": `Bearer ${env.RESEND_API_KEY}`, "Content-Type": "application/json", "Accept": "application/json" }, body: JSON.stringify( resendPayload ) } ); const responseText = await resendResponse.text(); let resendResult = null; if ( responseText .trim() .length ) { try { resendResult = JSON.parse( responseText ); } catch { resendResult = { raw: responseText }; } } /* * -------------------------------------------------------- * RESEND ERROR * -------------------------------------------------------- */ if ( !resendResponse.ok ) { console.error( "[CONTACT] Resend API failed:", resendResult ); return jsonResponse( { success: false, error: resendResult?.message || resendResult?.error || "Email service failed to send the message." }, corsHeaders, 502 ); } /* * -------------------------------------------------------- * SUCCESS * -------------------------------------------------------- */ console.log( "[CONTACT] Message sent successfully.", { siteName, email, hasAttachment: Boolean( attachment ), resendId: resendResult?.id || null } ); return jsonResponse( { success: true, message: "Email sent successfully.", id: resendResult?.id || null }, corsHeaders ); } catch ( error ) { console.error( "[CONTACT] Unexpected Worker error:", error ); return jsonResponse( { success: false, error: error?.message || "Unable to process your message." }, corsHeaders, 500 ); } } }; /* ============================================================================ REQUEST PARSERS ========================================================================== */ async function parseMultipartRequest( request, env ) { const formData = await request.formData(); const standardFields = new Set([ "name", "email", "siteName", "message", "topic", "order_id", "orderId", "source", "_gotcha", "attachment" ]); const extraFields = {}; for ( const [ key, value ] of formData.entries() ) { if ( standardFields.has( key ) ) { continue; } /* * Ignore files in generic extra-field collection. */ if ( value instanceof File ) { continue; } extraFields[key] = cleanText( value ); } const attachmentValue = formData.get( "attachment" ); return { name: cleanText( formData.get( "name" ) ), email: cleanText( formData.get( "email" ) ), siteName: cleanText( formData.get( "siteName" ) ), message: cleanText( formData.get( "message" ) ), topic: cleanText( formData.get( "topic" ) ), orderId: cleanText( formData.get( "order_id" ) || formData.get( "orderId" ) ), source: cleanText( formData.get( "source" ) ), honeypot: cleanText( formData.get( "_gotcha" ) ), attachment: attachmentValue instanceof File ? attachmentValue : null, extraFields }; } async function parseJsonRequest( request ) { let data; try { data = await request.json(); } catch { throw new Error( "Invalid JSON request." ); } if ( !data || typeof data !== "object" ) { throw new Error( "JSON request must contain an object." ); } const standardFields = new Set([ "name", "email", "siteName", "message", "topic", "order_id", "orderId", "source", "_gotcha" ]); const extraFields = {}; for ( const [ key, value ] of Object.entries( data ) ) { if ( standardFields.has( key ) ) { continue; } if ( value && typeof value === "object" ) { continue; } extraFields[key] = cleanText( value ); } return { name: cleanText( data.name ), email: cleanText( data.email ), siteName: cleanText( data.siteName ), message: cleanText( data.message ), topic: cleanText( data.topic ), orderId: cleanText( data.order_id || data.orderId ), source: cleanText( data.source ), honeypot: cleanText( data._gotcha ), attachment: null, extraFields }; } async function parseUrlEncodedRequest( request ) { const formData = await request.formData(); const standardFields = new Set([ "name", "email", "siteName", "message", "topic", "order_id", "orderId", "source", "_gotcha" ]); const extraFields = {}; for ( const [ key, value ] of formData.entries() ) { if ( standardFields.has( key ) ) { continue; } extraFields[key] = cleanText( value ); } return { name: cleanText( formData.get( "name" ) ), email: cleanText( formData.get( "email" ) ), siteName: cleanText( formData.get( "siteName" ) ), message: cleanText( formData.get( "message" ) ), topic: cleanText( formData.get( "topic" ) ), orderId: cleanText( formData.get( "order_id" ) || formData.get( "orderId" ) ), source: cleanText( formData.get( "source" ) ), honeypot: cleanText( formData.get( "_gotcha" ) ), attachment: null, extraFields }; } /* ============================================================================ VALIDATION ========================================================================== */ function isValidEmail( value ) { const email = cleanText( value ); if ( !email || email.length > 254 ) { return false; } /* * Practical email validation. */ return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test( email ); } /* ============================================================================ FILE HELPERS ========================================================================== */ function getFileExtension( filename ) { const cleanName = cleanText( filename ) .toLowerCase(); const index = cleanName.lastIndexOf( "." ); if ( index < 0 || index === cleanName.length - 1 ) { return ""; } return cleanName.slice( index + 1 ); } function sanitizeFileName( filename ) { let name = cleanText( filename ); /* * Remove directory traversal. */ name = name.replace( /[\\/]/g, "_" ); /* * Remove control characters. */ name = name.replace( /[\u0000-\u001F\u007F]/g, "" ); /* * Keep a reasonable filename length. */ if ( name.length > 180 ) { const extension = getFileExtension( name ); if ( extension ) { name = `${name.slice( 0, 160 )}.${extension}`; } else { name = name.slice( 0, 180 ); } } return ( name || "attachment" ); } /* ============================================================================ BASE64 ========================================================================== */ function uint8ArrayToBase64( bytes ) { let binary = ""; const chunkSize = 0x8000; for ( let i = 0; i < bytes.length; i += chunkSize ) { const chunk = bytes.subarray( i, Math.min( i + chunkSize, bytes.length ) ); binary += String.fromCharCode( ...chunk ); } return btoa( binary ); } /* ============================================================================ TEXT ========================================================================== */ function cleanText( value ) { if ( value === null || value === undefined ) { return ""; } return String( value ) .replace( /\s+/g, " " ) .trim(); } function formatFieldName( key ) { return String( key || "" ) .replace( /[_-]+/g, " " ) .replace( /\b\w/g, letter => letter.toUpperCase() ) .trim(); } function sanitizeSenderName( value ) { const result = cleanText( value ) .replace( /[\r\n<>]/g, "" ); return ( result || "Website" ) .slice( 0, 100 ); } /* ============================================================================ CORS ========================================================================== */ function getCorsHeaders( request ) { const origin = request.headers.get( "Origin" ); const allowedOrigins = [ "https://shop.prasunbarua.com", "https://prasunbarua.com", "https://www.prasunbarua.com" ]; /* * Only reflect known production origins. */ const allowOrigin = origin && allowedOrigins.includes( origin ) ? origin : "https://shop.prasunbarua.com"; return { "Access-Control-Allow-Origin": allowOrigin, "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Accept", "Access-Control-Max-Age": "86400", "Vary": "Origin" }; } /* ============================================================================ JSON RESPONSE ========================================================================== */ function jsonResponse( data, headers, status = 200 ) { return new Response( JSON.stringify( data ), { status, headers: { ...headers, "Content-Type": "application/json; charset=UTF-8", "Cache-Control": "no-store" } } ); }
